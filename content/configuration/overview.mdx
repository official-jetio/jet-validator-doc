# Configuration Overview

## Configuration Options

### Constructor

```typescript
import { JetValidator } from "@jetio/validator";

const jetValidator = new JetValidator(options);
```

### Available Options

@jetio/validator accepts a configuration object with the following options. All options are optional and have sensible defaults.

```typescript
interface ValidatorOptions {
  // debugging
  logFunction?: boolean;

  // code generation
  loopEnum?: number;
  loopRequired?: number;

  // $ref options
  draft?: "draft2019-09" | "draft2020-12" | "draft7" | "draft6";
  inlineRefs?: boolean;
  debug?: boolean;

  // Error Handling
  allErrors?: boolean;
  verbose?: boolean;

  // Validation Strictness
  strict?: boolean;
  strictNumbers?: boolean;
  strictRequired?: boolean;
  strictTypes?: boolean | "log";
  strictSchema?: boolean;

  // Data Modification
  coerceTypes?: boolean | "array";
  removeAdditional?: boolean | "all";
  useDefaults?: boolean | "empty";

  // Performance
  cache?: boolean;

  // Metaschema & Schema
  metaSchema?: string;
  validateSchema?: boolean;
  addUsedSchema?: boolean;
  loadSchema?: (uri: string) => Promise<SchemaDefinition> | SchemaDefinition;

  // Custom errors
  errorMessage?: boolean;

  // Advanced
  $data?: boolean;
  async?: boolean;

  // Formats
  validateFormats?: boolean;
  allowFormatOverride?: boolean;
  formatMode?: "full" | false | "fast";
  formats?: string[];
  overwrittenFormats?: string[]; // standalone code generation
}
```
